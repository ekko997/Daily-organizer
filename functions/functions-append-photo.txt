exports.parseEventPhoto = onCall({ secrets: [anthropicApiKey] }, async (request) => {
  const imageBase64 = request.data && request.data.imageBase64;
  const mimeType = (request.data && request.data.mimeType) || "image/jpeg";
  const localNow = request.data && request.data.localNow;
  const timeZone = request.data && request.data.timeZone;
  if (!imageBase64) {
    throw new Error("Missing image to parse");
  }

  const effectiveNow = localNow || new Date().toISOString().slice(0, 19);
  const dayOfWeek = new Date(effectiveNow).toLocaleDateString("en-US", { weekday: "long" });

  const prompt = `You are a warm, sharp personal assistant helping someone add something to their calendar from a photo — a flyer, invitation, school note, or similar. Think of a friend who's great at organizing, not a form-filling robot. The person's current local date and time is ${effectiveNow}${timeZone ? ` (their timezone: ${timeZone})` : ""} — that's a ${dayOfWeek}.

Look at the photo and figure out the event it describes — the title, when it is (resolving any relative dates against their current time above), and where if mentioned. If a time isn't stated, make a reasonable guess based on the type of event (default 9am for morning-sounding things, 7pm for evening-sounding things) — never leave it blank.

Respond with ONLY valid JSON, no other text, no markdown fences, in exactly this shape:
{"title": "short event title", "date": "YYYY-MM-DDTHH:MM:SS in their local time, in the future", "isAllDay": boolean, "category": one of "work","meeting","appointment","birthday","anniversary","personal","reminder", "location": "string or null", "confirmation": "a short, warm, one-sentence confirmation of what you understood from the photo, sound like a helpful person not a computer, under 20 words"}

If the photo genuinely doesn't show anything resembling an event, respond with exactly: {"error": "I couldn't find event details in that photo"}`;

  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "x-api-key": anthropicApiKey.value(),
      "anthropic-version": "2023-06-01",
    },
    body: JSON.stringify({
      model: "claude-haiku-4-5-20251001",
      max_tokens: 500,
      messages: [
        {
          role: "user",
          content: [
            { type: "image", source: { type: "base64", media_type: mimeType, data: imageBase64 } },
            { type: "text", text: prompt },
          ],
        },
      ],
    }),
  });

  if (!response.ok) {
    const errText = await response.text();
    throw new Error(`AI request failed: ${errText}`);
  }

  const data = await response.json();
  const rawText = (data.content && data.content[0] && data.content[0].text) || "{}";
  const cleaned = rawText.replace(/```json|```/g, "").trim();

  let parsed;
  try {
    parsed = JSON.parse(cleaned);
  } catch (e) {
    throw new Error("Could not read that photo — try a clearer shot");
  }

  if (parsed.error) {
    throw new Error(parsed.error);
  }

  return parsed;
});
