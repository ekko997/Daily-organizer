const { onDocumentWritten } = require("firebase-functions/v2/firestore");
const { onCall } = require("firebase-functions/v2/https");
const { defineSecret } = require("firebase-functions/params");
const admin = require("firebase-admin");
admin.initializeApp();

const EXPO_PUSH_URL = "https://exp.host/--/api/v2/push/send";
const anthropicApiKey = defineSecret("ANTHROPIC_API_KEY");

async function getMemberName(uid) {
  const snap = await admin.firestore().collection("users").doc(uid).get();
  if (!snap.exists) return "Someone";
  const data = snap.data();
  return data.displayName || (data.email ? data.email.split("@")[0] : "Someone");
}

async function sendPushToMembers(memberUids, excludeUid, title, body) {
  const db = admin.firestore();
  const tokens = [];
  for (const uid of memberUids) {
    if (uid === excludeUid) continue;
    const snap = await db.collection("users").doc(uid).get();
    const token = snap.exists ? snap.data().pushToken : null;
    if (token) tokens.push(token);
  }
  if (tokens.length === 0) return;

  const messages = tokens.map((to) => ({ to, title, body, sound: "default" }));
  await fetch(EXPO_PUSH_URL, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(messages),
  });
}

exports.onEventChange = onDocumentWritten("events/{eventId}", async (event) => {
  const change = event.data;
  const before = change.before && change.before.exists ? change.before.data() : null;
  const after = change.after && change.after.exists ? change.after.data() : null;
  const evt = after || before;
  if (!evt || evt.scope !== "family" || !evt.familyId) return;

  const familySnap = await admin.firestore().collection("families").doc(evt.familyId).get();
  if (!familySnap.exists) return;
  const members = familySnap.data().members || [];

  const authorUid = evt.lastModifiedBy;
  const authorName = authorUid ? await getMemberName(authorUid) : "Someone";
  const familyName = familySnap.data().name || "Family calendar";

  let body;
  if (!before && after) {
    body = `${authorName} added "${evt.title}"`;
  } else if (before && !after) {
    body = `"${evt.title}" was removed`;
  } else {
    body = `${authorName} updated "${evt.title}"`;
  }

  await sendPushToMembers(members, authorUid, familyName, body);
});

exports.onTodoChange = onDocumentWritten("todos/{todoId}", async (event) => {
  const change = event.data;
  const before = change.before && change.before.exists ? change.before.data() : null;
  const after = change.after && change.after.exists ? change.after.data() : null;
  const todo = after || before;
  if (!todo || todo.scope !== "family" || !todo.familyId) return;

  const familySnap = await admin.firestore().collection("families").doc(todo.familyId).get();
  if (!familySnap.exists) return;
  const members = familySnap.data().members || [];

  const authorUid = todo.lastModifiedBy;
  const authorName = authorUid ? await getMemberName(authorUid) : "Someone";
  const familyName = familySnap.data().name || "Family to-do";

  let body;
  if (!before && after) {
    body = `${authorName} added "${todo.text}"`;
  } else if (before && !after) {
    body = `"${todo.text}" was removed`;
  } else if (before && after && before.done !== after.done && after.done) {
    body = `${authorName} checked off "${todo.text}"`;
  } else {
    return;
  }

  await sendPushToMembers(members, authorUid, familyName, body);
});

// Parses a natural-language request ("dentist next Tuesday 3pm") into
// structured event fields using Claude. The API key lives here, server-side,
// as a Firebase secret — never shipped inside the app itself.
exports.parseEventText = onCall({ secrets: [anthropicApiKey] }, async (request) => {
  const text = request.data && request.data.text;
  if (!text || typeof text !== "string") {
    throw new Error("Missing text to parse");
  }

  const now = new Date();
  const prompt = `You are helping parse a natural-language request into a calendar event. The current date and time is ${now.toISOString()} (day of week: ${now.toLocaleDateString("en-US", { weekday: "long" })}).

Parse this request into an event: "${text}"

Respond with ONLY valid JSON, no other text, no markdown code fences, in exactly this shape:
{"title": "short event title", "date": "ISO 8601 datetime string, in the future relative to now", "isAllDay": boolean, "category": one of "work","meeting","appointment","birthday","anniversary","personal","reminder", "location": "string or null"}`;

  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "x-api-key": anthropicApiKey.value(),
      "anthropic-version": "2023-06-01",
    },
    body: JSON.stringify({
      model: "claude-haiku-4-5-20251001",
      max_tokens: 300,
      messages: [{ role: "user", content: prompt }],
    }),
  });

  if (!response.ok) {
    const errText = await response.text();
    throw new Error(`AI request failed: ${errText}`);
  }

  const data = await response.json();
  const rawText = (data.content && data.content[0] && data.content[0].text) || "{}";
  const cleaned = rawText.replace(/```json|```/g, "").trim();

  let parsed;
  try {
    parsed = JSON.parse(cleaned);
  } catch (e) {
    throw new Error("Could not understand that — try rephrasing");
  }

  return parsed;
});
